// Fotos zu den Meisterschaftsergebnissen im Terminplaner
// In die vorhandenen Firestore-Regeln unter match /databases/{database}/documents einfügen.
// Lesen: alle (wie der Terminplaner). Schreiben: angemeldete Benutzer, nur Bilder bis ca. 900 KB.
match /meisterschaftsFotos/{terminId} {
  allow read: if true;
  allow delete: if request.auth != null;
  allow create, update: if request.auth != null
    && request.resource.data.keys().hasOnly(['data', 'eventId', 'name', 'date', 'updatedAt'])
    && request.resource.data.data is string
    && request.resource.data.data.size() < 950000
    && request.resource.data.eventId == terminId;
}
