// Laufende Wettbewerbe für "Mein Überblick" (Smartphone Mitglieder)
// In die vorhandenen Firestore-Regeln unter match /databases/{database}/documents einfügen.
// isAdmin() stammt aus den bestehenden Projektregeln.
match /smartphoneBetrieb/wettbewerbe {
  allow read: if true;
  allow write: if request.auth != null && isAdmin()
    && request.resource.data.keys().hasOnly([
      'duoCup', 'frauencup', 'montagsSolo', 'jahresmeisterschaft', 'geaendertAm'
    ])
    && request.resource.data.duoCup is bool
    && request.resource.data.frauencup is bool
    && request.resource.data.montagsSolo is bool
    && request.resource.data.jahresmeisterschaft is bool
    && request.resource.data.geaendertAm is string;
}
